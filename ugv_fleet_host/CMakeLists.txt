cmake_minimum_required(VERSION 3.16)
project(ugv_fleet_host VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Eigen3 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONCPP REQUIRED IMPORTED_TARGET jsoncpp)
find_package(Threads REQUIRED)
set(_ugv_ros_default OFF)
if(DEFINED CATKIN_DEVEL_PREFIX)
  set(_ugv_ros_default ON)
endif()
option(UGV_FLEET_WITH_ROS "Build the native fleet host with existing controller ROS data edges" ${_ugv_ros_default})
find_package(XgcXrpc REQUIRED CONFIG COMPONENTS http bootstrap)
add_library(ugv_fleet_reset_core STATIC
  ../ugv_reset_safety/src/reset_geometry.cpp
  ../ugv_reset_safety/src/reset_path.cpp
  ../ugv_reset_safety/src/fleet_coordinator.cpp)
target_include_directories(ugv_fleet_reset_core PUBLIC ../ugv_reset_safety/include include)
target_link_libraries(ugv_fleet_reset_core PUBLIC Eigen3::Eigen)
target_compile_options(ugv_fleet_reset_core PRIVATE -Wall -Wextra -Werror)
add_library(ugv_fleet_reset_http STATIC src/reset_http_boundary.cpp src/scene_cache.cpp)
target_link_libraries(ugv_fleet_reset_http PUBLIC ugv_fleet_reset_core XgcXrpc::http PkgConfig::JSONCPP Threads::Threads)
target_compile_definitions(ugv_fleet_reset_http PRIVATE BOOST_BIND_GLOBAL_PLACEHOLDERS)
target_compile_options(ugv_fleet_reset_http PRIVATE -Wall -Wextra -Werror)
if(NOT UGV_FLEET_WITH_ROS OR CATKIN_ENABLE_TESTING)
  enable_testing()
  find_package(GTest REQUIRED)
  foreach(case reset_control reset_http_boundary fixed_executor scene_cache fleet_runtime)
    add_executable(${case}_test test/${case}_test.cpp)
    target_link_libraries(${case}_test PRIVATE ugv_fleet_reset_http GTest::gtest_main)
    add_test(NAME ${case} COMMAND ${case}_test)
    set_tests_properties(${case} PROPERTIES TIMEOUT 15)
  endforeach()
  foreach(case fleet_coordinator reset_path reset_dwa reset_closed_loop reset_scenarios reset_session fleet_schedule)
    add_executable(ugv_fleet_${case}_test ../ugv_reset_safety/test/${case}_test.cpp)
    target_link_libraries(ugv_fleet_${case}_test PRIVATE ugv_fleet_reset_core GTest::gtest_main)
    add_test(NAME ${case} COMMAND ugv_fleet_${case}_test)
    set_tests_properties(${case} PROPERTIES TIMEOUT 30)
  endforeach()
  # All 12 production-profile scenarios share one CTest process.
  set_tests_properties(reset_scenarios PROPERTIES TIMEOUT 120)
endif()

  target_link_libraries(ugv_native_mecanum_core PUBLIC ugv_native_state_machine ugv_fleet_reset_core)

  target_link_libraries(ugv_native_unicycle_core PUBLIC ugv_native_state_machine ugv_fleet_reset_core
    ${UGV_NATIVE_ACADOS_PREFIX}/lib/libacados.so ${UGV_NATIVE_ACADOS_PREFIX}/lib/libhpipm.so
    ${UGV_NATIVE_ACADOS_PREFIX}/lib/libblasfeo.so m)

set(UGV_NATIVE_ROS_PREFIX "" CACHE PATH "Existing native Noetic ROS prefix for isolated edge/host validation")
if(UGV_NATIVE_ACADOS_PREFIX AND UGV_NATIVE_ROS_PREFIX)
  find_package(PkgConfig REQUIRED)
  set(ENV{PKG_CONFIG_PATH} "${UGV_NATIVE_ROS_PREFIX}/lib/pkgconfig:$ENV{PKG_CONFIG_PATH}")
  pkg_check_modules(UGV_ROSCPP REQUIRED IMPORTED_TARGET roscpp)
  set(UGV_NATIVE_LOG4CXX_LIBRARY "" CACHE FILEPATH "Existing log4cxx SONAME library when native ROS pkg-config points at a missing development symlink")
  if(UGV_NATIVE_LOG4CXX_LIBRARY)
    if(NOT EXISTS "${UGV_NATIVE_LOG4CXX_LIBRARY}")
      message(FATAL_ERROR "Explicit native log4cxx library is absent")
    endif()
    foreach(_ugv_ros_property INTERFACE_LINK_LIBRARIES INTERFACE_LINK_OPTIONS)
      get_target_property(_ugv_ros_libraries PkgConfig::UGV_ROSCPP ${_ugv_ros_property})
      if(_ugv_ros_libraries)
        list(TRANSFORM _ugv_ros_libraries REPLACE "^/usr/lib/x86_64-linux-gnu/liblog4cxx\\.so$" "${UGV_NATIVE_LOG4CXX_LIBRARY}")
        set_property(TARGET PkgConfig::UGV_ROSCPP PROPERTY ${_ugv_ros_property} "${_ugv_ros_libraries}")
      endif()
    endforeach()
  endif()
  add_library(ugv_native_mecanum_edge ${_ugv_mec}/src/mecanum_ugv_ros_node.cpp)
  set_property(TARGET ugv_native_mecanum_edge PROPERTY CXX_STANDARD 17)
  target_link_libraries(ugv_native_mecanum_edge PUBLIC ugv_native_mecanum_core PkgConfig::UGV_ROSCPP)
  add_library(ugv_native_unicycle_edge ${_ugv_uni}/src/ros_log_sink.cpp ${_ugv_uni}/src/unicycle_ugv_ros_node.cpp
    ${_ugv_uni}/src/input/command_input_producer.cpp ${_ugv_uni}/src/input/pva_reference_input_producer.cpp
    ${_ugv_uni}/src/input/reference_input_producer.cpp ${_ugv_uni}/src/input/reset_target_input_producer.cpp
    ${_ugv_uni}/src/input/state_input_producer.cpp ${_ugv_uni}/src/output/cmd_vel_output_consumer.cpp
    ${_ugv_uni}/src/output/nmpc_output_consumer.cpp)
  set_property(TARGET ugv_native_unicycle_edge PROPERTY CXX_STANDARD 17)
  target_include_directories(ugv_native_unicycle_edge PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../target/generated/include)
  target_link_libraries(ugv_native_unicycle_edge PUBLIC ugv_native_unicycle_core PkgConfig::UGV_ROSCPP)

  target_link_libraries(ugv_fleet_host_native PRIVATE ugv_fleet_reset_http ugv_native_mecanum_edge ugv_native_unicycle_edge)
endif()
